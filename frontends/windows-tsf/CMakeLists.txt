cmake_minimum_required(VERSION 3.20)
project(shixin_tsf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
add_executable(shixin_client_test test/core_test.cc)
target_include_directories(shixin_client_test PRIVATE src)
target_link_libraries(shixin_client_test PRIVATE Threads::Threads)
target_compile_definitions(shixin_client_test PRIVATE SHIXIN_GOLDEN="${CMAKE_CURRENT_SOURCE_DIR}/../../internal/ipc/testdata/golden.txt")
if(MSVC)
  target_compile_options(shixin_client_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(shixin_client_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME shixin_client_test COMMAND shixin_client_test)
if(WIN32)
  add_library(shixin_tsf SHARED src/dll.cc src/service.cc src/transport.cc src/register.cc)
  set_target_properties(shixin_tsf PROPERTIES PREFIX "")
  target_compile_definitions(shixin_tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(shixin_tsf PRIVATE ole32 oleaut32 uuid bcrypt advapi32 shell32)
  add_executable(shixin_dll_test test/dll_test.cc)
  target_include_directories(shixin_dll_test PRIVATE src)
  target_compile_definitions(shixin_dll_test PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(shixin_dll_test PRIVATE ole32 uuid)
  if(MINGW)
    target_link_options(shixin_dll_test PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME shixin_dll_test COMMAND shixin_dll_test $<TARGET_FILE:shixin_tsf>)
  if(MSVC)
    set_property(TARGET shixin_tsf PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(shixin_tsf PRIVATE /W4 /WX /utf-8 /guard:cf)
    target_link_options(shixin_tsf PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  else()
    target_compile_options(shixin_tsf PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
    target_link_options(shixin_tsf PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
